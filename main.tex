% --- INTRODUCTION ----------------------------------------------------------
\block{Introduction}{
\textbf{TartuBid} is a distributed online auction platform. Bidders
on the public Internet place bids on items hosted by sharded
auction servers; whoever holds the highest bid \emph{strictly
before} the deadline wins.

\bigskip
\textbf{Fairness depends on whose timestamp is trusted.} A cheap
quartz oscillator drifts roughly $1\msec$ per $20$~s if
uncorrected~\cite{vansteen2017ds}; modest skew between auction
servers can flip the winner of a close auction:

\bigskip

\bigskip
With eu-west $+80\msec$ fast and us-east $-60\msec$ slow, a
na\"ive coordinator would \emph{reject} the on-time bid and
\emph{accept} the late one.

\bigskip
\textbf{An online auction therefore demands \emph{strict} clock
alignment around the deadline} --- not merely accurate-on-average.
We compare two standard baselines against an advanced mechanism
that meets this stricter requirement.
}

% --- SYSTEM MODEL ----------------------------------------------------------
\block{System model}{

\textbf{Architecture.} Bidder clients issue bids over HTTPS to one
of several auction servers, each authoritative for a shard of
items. A central \emph{coordinator} resolves winners after each
item's deadline.

\bigskip
\textbf{Timing model.} Partially synchronous: bounded but unknown
delay and drift; cross-server skew is the dominant hazard near a
closing window.

\textbf{Failure model.} Crash and omission failures, masked by
retry + persistence. Byzantine failures are out of scope.
}

% --- MECHANISMS (compact 2-column inner layout per mechanism) --------------
\block{Mechanisms}{

% ======= (a) NTP =======
\begin{minipage}[t]{0.50\linewidth}\vspace{0pt}
\textbf{(a) NTP --- Network Time Protocol.}\\
Four-timestamp exchange against a stratum-$n$ reference; client
estimates offset $\theta$ and round-trip delay
$\delta$~\cite{mills1991ntp,rfc5905}.
\[
   \theta = \tfrac{(t_2 - t_1) + (t_3 - t_4)}{2},
   \;\;
   \delta = (t_4 - t_1) - (t_3 - t_2).
\]
\end{minipage}%
\hfill
\begin{minipage}[t]{0.46\linewidth}\vspace{0pt}
\centering
\begin{tikzpicture}[font=\large, x=1.3cm, y=1.0cm]
  \node[font=\large\bfseries] at (0.8, 4.4) {Client};
  \node[font=\large\bfseries] at (6.0, 4.4) {Reference};
  \draw[ultra thick, DarkBlue] (0.8, 3.8) -- (0.8, 0.0);
  \draw[ultra thick, DarkBlue] (6.0, 3.8) -- (6.0, 0.0);
  \draw[-{Stealth[length=3.5mm]}, very thick] (0.8, 3.2) -- (6.0, 2.6);
  \filldraw[DarkBlue] (0.8, 3.2) circle (3pt) node[left, font=\large] {$t_1$};
  \filldraw[DarkBlue] (6.0, 2.6) circle (3pt) node[right, font=\large] {$t_2$};
  \draw[-{Stealth[length=3.5mm]}, very thick] (6.0, 1.4) -- (0.8, 0.8);
  \filldraw[DarkBlue] (6.0, 1.4) circle (3pt) node[right, font=\large] {$t_3$};
  \filldraw[DarkBlue] (0.8, 0.8) circle (3pt) node[left, font=\large] {$t_4$};
\end{tikzpicture}
\end{minipage}

\smallskip
% ======= (b) Berkeley =======
\begin{minipage}[t]{0.50\linewidth}\vspace{0pt}
\textbf{(b) Berkeley algorithm.}\\
A daemon polls every server's clock, averages them, and sends each
server an individual adjustment~\cite{gusella1989berkeley}.
\[
   C_{\text{cons}} = \tfrac{1}{n}\sum_{i} C_i,
   \;\; \Delta_i = C_{\text{cons}} - C_i.
\]
\end{minipage}%
\hfill
\begin{minipage}[t]{0.46\linewidth}\vspace{0pt}
\centering

               ultra thick, align=center},

\end{tikzpicture}
\end{minipage}

\smallskip
% ======= (c) TrueTime =======
\begin{minipage}[t]{0.50\linewidth}\vspace{0pt}
\textbf{(c) TrueTime (Google Spanner).}\\
Returns a \emph{bounded} interval rather than a point; with
\emph{commit-wait}, ambiguous orderings are \textbf{refused}
rather than guessed~\cite{corbett2012spanner}.
\[
   \mathrm{TT.now}() = [\,t - \varepsilon,\; t + \varepsilon\,].
\]
\end{minipage}%
\hfill
\begin{minipage}[t]{0.46\linewidth}\vspace{0pt}

% --- COMPARISON AND OUR CHOICE --------------------------------------------
\block{Comparison and our choice}{

\renewcommand{\arraystretch}{1.35}
\begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}p{6.6cm} C C C@{}}
\toprule
                          & \textbf{NTP}              & \textbf{Berkeley}     & \textbf{TrueTime}         \\
\midrule
Typical accuracy          & \ycell{$1\text{--}10\msec$ (WAN)} & \ycell{ms (internal)} & \gcell{$\varepsilon\!\le\!7\msec$} \\
External UTC alignment    & \gcell{yes}               & \rcell{no}            & \gcell{yes}               \\
Extra hardware            & \gcell{none}              & \gcell{none}          & \rcell{GPS + atomic / DC} \\
Handles unknown delay     & \ycell{via filtering}     & \ycell{via averaging} & \gcell{via $\varepsilon$ bound} \\
Refuses ambiguous order   & \rcell{no}                & \rcell{no}            & \gcell{yes (commit-wait)} \\
Operational cost          & \gcell{low}               & \gcell{very low}      & \rcell{high}              \\
Failure if reference down & \ycell{free-run drift}    & \gcell{still converges} & \ycell{widens $\varepsilon$} \\
\bottomrule
\end{tabularx}

\bigskip
\raggedright
\textbf{Justification.}
\emph{Berkeley alone} converges the cluster on \emph{some} time,
but not on UTC, and bidders' own clocks (UI countdowns) are
excluded from the average.
\emph{NTP} delivers $\sim$millisecond accuracy at zero marginal
cost --- enough for our deadline granularity.

\bigskip
\centering
\colorbox{Blue10}{\parbox{0.95\linewidth}{\centering\vspace{6pt}
\textbf{We adopt NTP everywhere, with a TrueTime-inspired
uncertainty band $\varepsilon = 50\msec$ around each closing
instant.}\\[6pt]
\large Bids inside
$[\,t_{\text{close}}-\varepsilon,\;t_{\text{close}}+\varepsilon\,]$
trigger \emph{commit-wait} --- NTP's cost profile with TrueTime's
safety, exactly where it matters.
\vspace{6pt}}}
}

% --- PROTOTYPE EVIDENCE ----------------------------------------------------
\block{Prototype evidence}{
Python prototype (\texttt{tartubid/}, $\sim$300 lines) replays the
Introduction scenario with $\pm 80\msec$ deliberate skew.

\bigskip
\renewcommand{\arraystretch}{1.4}
\begin{tabularx}{\linewidth}{@{}L L L@{}}
\toprule
\textbf{Strategy}            & \textbf{Resolved winner}         & \textbf{Outcome} \\
\midrule
Naive (no sync)              & Mumin (110)                      & \rcell{$\times$\;\textbf{wrong winner}} \\
NTP                          & Anup (100)                       & \gcell{$\checkmark$\;\textbf{matched, no guarantee}} \\
Berkeley                     & Anup (100)                       & \gcell{$\checkmark$\;\textbf{matched, no guarantee}} \\
TrueTime                     & ---~commit-wait~---              & \cellcolor{Blue20}\textbf{safe (no premature commit)} \\
\bottomrule
\end{tabularx}

\bigskip
\raggedright
The na\"ive resolver hands the auction to the wrong bidder. NTP
and Berkeley happen to match the true winner in this run, but
neither refuses ambiguous orderings --- the match is run-specific,
not guaranteed. TrueTime alone refuses to commit while the
deadline lies inside the uncertainty interval.
}
